# The upstream helper also installs binaries. Apply its Windows DLL properties to this isolated target.
function(set_target_properties_obs target)
  set_target_properties(${target} ${ARGN})
  if(target STREQUAL libobs)
    set_target_properties(libobs PROPERTIES VERSION "${OBS_VERSION_MAJOR}" SOVERSION "${OBS_VERSION_CANONICAL}")
    target_link_options(libobs PRIVATE "/PDBALTPATH:obs.pdb")
  endif()
endfunction()
function(target_export target)
endfunction()
